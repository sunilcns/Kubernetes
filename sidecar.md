# product-with-sidecar.yaml
apiVersion: v1
kind: Pod
metadata:
  name: product-with-sidecar
  labels:
    app: product
    project: shopeasy
spec:
  containers:
    - name: product            # Main container
      image: nginx
      ports:
        - containerPort: 80
      volumeMounts:
        - name: shared-logs
          mountPath: /var/log/nginx

    - name: log-shipper        # Sidecar container
      image: busybox
      command: ["sh", "-c", "while true; do echo '[LOG-SHIPPER] Sending logs to CloudWatch...'; sleep 5; done"]
      volumeMounts:
        - name: shared-logs
          mountPath: /logs

  volumes:
    - name: shared-logs        # Shared volume between both containers
      emptyDir: {}

# | Container     | Purpose                       |
# | ------------- | ----------------------------- |
# | `product`     | Runs main application (nginx) |
# | `log-shipper` | Collects/sends logs           |


# log-shipper is a sidecar container
# A sidecar container runs alongside the main application container inside the same Pod.  It simulates a logging agent.
# Main container → product (nginx app) and Sidecar container → log-shipper

# The sidecar’s job is usually:
# collect logs, monitor files, forward logs to external systems Read application logs and Send logs to AWS CloudWatch, ES




# In this Kubernetes Pod, the `log-shipper` acts as a sidecar container that collects logs generated by the main application container (`product`).
# Both containers share a common `emptyDir` volume, where the application writes log files and the sidecar reads them for 
# forwarding to external logging systems such as CloudWatch, Elasticsearch, or Loki. Volume mounts are important here because 
# containers inside a Pod have separate filesystems by default, and the shared volume allows them to exchange log data. 
# The `emptyDir` volume is temporary storage that exists only as long as the Pod is running. If the Pod is deleted, 
# all data inside the shared volume is permanently removed.

# Main Container
# mountPath: /var/log/nginx
# Nginx logs get stored here.
# Sidecar Container
# mountPath: /logs
# The same shared volume appears inside sidecar as /logs.
# So sidecar can access nginx logs.
# Shared Volume
# volumes:
#   - name: shared-logs
#     emptyDir: {}
# emptyDir creates temporary storage shared between containers in same Pod.Visible to: product container & log-shipper container
# as  both are inside the same Pod.Another Pod CANNOT access this volume
# If You Want Multiple Pods to Share Data  Use: PersistentVolume (PV) PersistentVolumeClaim (PVC) NFS

# When product container writes logs to /var/log/nginx, they are actually stored in the shared volume. without volume mount sidecar cannot read nginx logs
# The log-shipper sidecar can read those logs from /logs and forward them to external logging systems.
